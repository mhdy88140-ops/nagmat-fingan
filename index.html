// نموذج لبيانات المنتجات في متجرك (يمكنك تعديلها وإضافة منتجاتك) const products = [ { id: 1, name: "مطحنة يوية 1", category: "المطاحن اليدوية", price: "25,000 د.ع", image: "img1.jpg" }, { id: 2, name: "قمع تقطير V60 زجاجي", category: "ادوات تقطير", price: "18,000 د.ع", image: "img2.jpg" }, { id: 3, name: "فلتر V60 ورقي", category: "ادوات تقطير", price: "5,000 د.ع", image: "img3.jpg" }, { id: 4, name: "تامبر قهوة اسبريسو", category: "التامبر والادوات", price: "15,000 د.ع", image: "img4.jpg" } ]; // دالة تصفية المنتجات عند الضغط على القسم (مثل المطاحن أو V60) function filterByCategory(categoryName) { const filtered = products.filter(p => p.category.includes(categoryName) || p.name.includes(categoryName)); displayProducts(filtered); } // دالة البحث function searchProducts(query) { const searchTerm = query.toLowerCase(); const results = products.filter(p => p.name.toLowerCase().includes(searchTerm) || p.category.toLowerCase().includes(searchTerm)); displayProducts(results); } // دالة لعرض النتائج في واجهة المتجر (تأكد من مطابقة اسم الـ ID لعنصر العرض لديك) function displayProducts(items) { const container = document.getElementById('products-container'); // ضع هنا الـ ID الخاص بحاوية المنتجات في موقعك if (!container) return; container.innerHTML = ''; if (items.length === 0) { container.innerHTML = '

لا توجد نتائج مطابقة لبحثك.

'; return; } items.forEach(product => { container.innerHTML += `

${product.name}

السعر: ${product.price}

`; }); } // ربط الأحداث بأزرار القائمة وشريط البحث document.addEventListener('DOMContentLoaded', () => { // ربط روابط الأقسام في القائمة الجانبية document.querySelectorAll('.menu-category-item').forEach(item => { item.addEventListener('click', (e) => { const category = e.target.getAttribute('data-category'); filterByCategory(category); }); }); // ربط شريط البحث const searchInput = document.getElementById('search-input'); // ضع الـ ID الخاص بحقل البحث لديك if (searchInput) { searchInput.addEventListener('input', (e) => { searchProducts(e.target.value); }); } });
